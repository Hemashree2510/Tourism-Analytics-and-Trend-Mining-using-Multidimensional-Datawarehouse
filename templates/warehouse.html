Data Warehouse Architecture & Schema

Multidimensional Star Schema Design & ETL Pipeline
Grain: Country x Year x Tourism Type

The TourismVista Data Warehouse implements an academic Star Schema architecture designed to support high-performance On-Line Analytical Processing (OLAP) and Data Mining queries. The central fact_tourism table records international tourist arrival measures connected to four specialized dimension tables.

Data Definition Language (DDL) Script

SQL Standard / Oracle / MySQL Compatible 
-- DIM_COUNTRY
CREATE TABLE dim_country(
    country_key INT PRIMARY KEY,
    country_name VARCHAR(100) NOT NULL
);

-- DIM_TIME
CREATE TABLE dim_time(
    time_key INT PRIMARY KEY,
    year_num INT NOT NULL,
    year_category VARCHAR(30) NOT NULL
);

-- DIM_TOURISMTYPE
CREATE TABLE dim_tourism_type(
    tourism_type_key INT PRIMARY KEY,
    data_type VARCHAR(50) NOT NULL,
    actual_or_predicted VARCHAR(20) NOT NULL,
    data_source VARCHAR(100) NOT NULL
);

-- DIM_WORLDBANK
CREATE TABLE dim_worldbank(
    worldbank_key INT PRIMARY KEY,
    worldbank_data_year INT NOT NULL,
    source_name VARCHAR(100) NOT NULL
);

-- FACT_TOURISM
CREATE TABLE fact_tourism(
    tourism_key INT PRIMARY KEY,
    country_key INT,
    time_key INT,
    tourism_type_key INT,
    worldbank_key INT,
    tourist_arrivals_millions DECIMAL(12,3),
    CONSTRAINT fk_tourism_country FOREIGN KEY (country_key) REFERENCES dim_country(country_key),
    CONSTRAINT fk_tourism_time FOREIGN KEY (time_key) REFERENCES dim_time(time_key),
    CONSTRAINT fk_tourism_type FOREIGN KEY (tourism_type_key) REFERENCES dim_tourism_type(tourism_type_key),
    CONSTRAINT fk_tourism_worldbank FOREIGN KEY (worldbank_key) REFERENCES dim_worldbank(worldbank_key)
);
